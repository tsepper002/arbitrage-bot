```python
class MultiHopEngine:
```

```python
# core/gas_oracle.py
class GasOracle:
    def __init__(self):
        self.sources = {
            'ethereum': 'https://api.etherscan.io/api?module=gastracker',
            'arbitrum': 'https://api.arbiscan.io/api?module=gastracker',
            'polygon': 'https://api.polygonscan.com/api?module=gastracker'
        }
        
    async def get_current_fee(self, network, currency='USDT'):
        """Получить текущую комиссию перевода"""
        gas_price = await self._fetch_gas_price(network)
        usdt_transfer_gas = 65000  # gas units для ERC20 transfer
        fee_eth = (gas_price * usdt_transfer_gas) / 1e9
        fee_usd = fee_eth * await self._get_eth_price()
        return fee_usd
        
    async def select_cheapest_network(self, amount):
        """Выбрать самую дешёвую сеть сейчас"""
        fees = {}
        for network in ['ERC20', 'Arbitrum', 'Polygon', 'BEP20']:
            fees[network] = await self.get_current_fee(network)
        return min(fees, key=fees.get)
```

```python
# core/market_maker.py
class MarketMaker:
    def place_maker_orders(self, symbol, exchange='MEXC'):
        """Разместить лимитные ордера на обеих сторонах спреда"""
        
        # Получить текущий спред
```

```python
# core/slippage_predictor.py
class SlippagePredictor:
    def __init__(self):
        self.history = []  # История исполнений
        
    def predict_slippage(self, symbol, exchange, side, quantity):
        """Предсказать slippage в %"""
        
        # Факторы
        orderbook_depth = self._get_depth_at_quantity(symbol, exchange, quantity)
        volatility = self._get_recent_volatility(symbol)
        time_of_day = datetime.now().hour
        
        # Исторический анализ
        similar_trades = self._find_similar_trades(
            symbol, exchange, quantity, time_of_day
        )
        avg_slippage = np.mean([t['actual_slippage'] for t in similar_trades])
        
        # Корректировка на текущие условия
        depth_factor = 1.0 if orderbook_depth > quantity * 2 else 1.5
        volatility_factor = 1.0 + volatility / 10
        
        predicted_slippage = avg_slippage * depth_factor * volatility_factor
        return predicted_slippage
        
    def should_execute(self, opportunity, predicted_slippage):
        """Исполнять только если slippage приемлем"""
        net_profit = opportunity['net_profit_pct'] - predicted_slippage
        return net_profit > settings.MIN_NET_ROI_PCT
```

### 7. Volume-Weighted Spread Analysis (2-3 ч)

**Текущая проблема**: Смотрим только на топ стакана
```python
# Сейчас:
best_bid = orderbook['bids'][0]  # Только топ
best_ask = orderbook['asks'][0]
```

**Решение**: Анализ кумулятивного объёма
```python
# core/depth_analyzer.py
class DepthAnalyzer:
    def get_executable_price(self, orderbook, side, quantity):
        """Получить реальную цену исполнения для quantity"""
        
        levels = orderbook['bids'] if side == 'buy' else orderbook['asks']
        cumulative_qty = 0
        weighted_price = 0
        
        for price, qty in levels:
            if cumulative_qty >= quantity:
                break
            
            # Сколько можем взять с этого уровня
            take_qty = min(qty, quantity - cumulative_qty)
            weighted_price += price * take_qty
            cumulative_qty += take_qty
            
        if cumulative_qty < quantity:
            return None  # Недостаточно ликвидности
            
        return weighted_price / cumulative_qty
        
    def get_true_spread(self, symbol, buy_exchange, sell_exchange, quantity):
        """Реальный спред с учётом объёма"""
        buy_price = self.get_executable_price(
            orderbooks[buy_exchange][symbol], 'buy', quantity
        )
        sell_price = self.get_executable_price(
            orderbooks[sell_exchange][symbol], 'sell', quantity
        )
        
        if not buy_price or not sell_price:
            return None
            
        spread_pct = (sell_price - buy_price) / buy_price * 100
        return spread_pct
```

**Ожидаемый эффект**:
- Более точные расчёты прибыли
- Меньше failed trades из-за недостаточной ликвидности
- +3-7% точность

### 9. Backtesting Engine (6-8 ч)

**Описание**: Тестировать стратегии на исторических данных

**Реализация**:
```python
# backtest/engine.py
class BacktestEngine:
    def __init__(self, historical_data_path):
        self.data = self._load_historical_orderbooks(historical_data_path)
        self.pnl = 0
        self.trades = []
        
    def run_backtest(self, start_date, end_date, strategy_params):
        """Запустить backtest"""
        
        # Симулировать время
        for timestamp in self.data.timestamps:
            if not (start_date <= timestamp <= end_date):
                continue
                
            # Получить orderbooks на этот момент
            orderbooks = self.data.get_orderbooks_at(timestamp)
            
            # Запустить стратегию
            opportunities = self.strategy.scan(orderbooks, strategy_params)
            
            # Симулировать исполнение
            for opp in opportunities:
                result = self._simulate_execution(opp, orderbooks)
                self.trades.append(result)
                self.pnl += result['profit']
                
        # Статистика
        return {
            'total_pnl': self.pnl,
            'num_trades': len(self.trades),
            'win_rate': self._calculate_win_rate(),
            'sharpe_ratio': self._calculate_sharpe(),
            'max_drawdown': self._calculate_max_drawdown()
        }
        
    def optimize_parameters(self):
        """Найти оптимальные параметры"""
        best_pnl = -float('inf')
        best_params = None
        
        # Grid search
        for min_roi in [0.02, 0.03, 0.05, 0.08]:
            for cooldown in [1, 3, 5, 10]:
                params = {'min_roi': min_roi, 'cooldown': cooldown}
                results = self.run_backtest(..., params)
                if results['total_pnl'] > best_pnl:
                    best_pnl = results['total_pnl']
                    best_params = params
                    
        return best_params
```

```python
# .env
ARB_BYBIT_KEY_1=xxx
ARB_BYBIT_SECRET_1=xxx
ARB_BYBIT_KEY_2=yyy
ARB_BYBIT_SECRET_2=yyy

# core/multi_account.py
class MultiAccountManager:
    def __init__(self):
        self.accounts = {
            'Bybit': [
                BybitRESTClient(key1, secret1),
                BybitRESTClient(key2, secret2)
            ],
            # ...
        }
        self.account_index = defaultdict(int)  # Round-robin
        
    def get_next_account(self, exchange):
        """Выбрать следующий аккаунт (round-robin)"""
        accounts = self.accounts[exchange]
        idx = self.account_index[exchange]
        self.account_index[exchange] = (idx + 1) % len(accounts)
        return accounts[idx]
        
    async def place_order_distributed(self, exchange, order):
        """Распределить ордер между аккаунтами"""
        account = self.get_next_account(exchange)
        return await account.place_order(order)
```
